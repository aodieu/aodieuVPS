function e($v) { return htmlspecialchars((string)$v, ENT_QUOTES, 'UTF-8'); }
function fmtbytes($size) {
    $units = array('B','KiB','MiB','GiB','TiB');
    $i = 0;
    while ($size >= 1024 && $i < count($units) - 1) { $size /= 1024; $i++; }
    return number_format($size, 2) . ' ' . $units[$i];
}
$status = function_exists('opcache_get_status') ? @opcache_get_status(false) : false;
$configuration = function_exists('opcache_get_configuration') ? @opcache_get_configuration() : false;
$statistics = $status && isset($status['opcache_statistics']) ? $status['opcache_statistics'] : array();
$memory = $status && isset($status['memory_usage']) ? $status['memory_usage'] : array();
$items = array(
    'Enabled' => $status && !empty($status['opcache_enabled']) ? 'Yes' : 'No',
    'OPcache extension' => extension_loaded('Zend OPcache') ? 'Installed' : 'Not installed',
    'PHP version' => PHP_VERSION,
    'Cache hits' => isset($statistics['hits']) ? number_format($statistics['hits']) : '—',
    'Cache misses' => isset($statistics['misses']) ? number_format($statistics['misses']) : '—',
    'Hit rate' => isset($statistics['opcache_hit_rate']) ? number_format($statistics['opcache_hit_rate'], 2) . '%' : '—',
    'Cached scripts' => isset($statistics['num_cached_scripts']) ? number_format($statistics['num_cached_scripts']) : '—',
    'Used memory' => isset($memory['used_memory']) ? fmtbytes($memory['used_memory']) : '—',
    'Free memory' => isset($memory['free_memory']) ? fmtbytes($memory['free_memory']) : '—',
    'Wasted memory' => isset($memory['wasted_memory']) ? fmtbytes($memory['wasted_memory']) : '—',
    'Restarts' => isset($statistics['oom_restarts']) ? number_format($statistics['oom_restarts'] + (isset($statistics['hash_restarts']) ? $statistics['hash_restarts'] : 0) + (isset($statistics['manual_restarts']) ? $statistics['manual_restarts'] : 0)) : '—',
);
